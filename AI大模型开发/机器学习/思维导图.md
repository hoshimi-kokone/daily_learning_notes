$$
\text{Machine Learning（机器学习）}
\left\{
\begin{array}{l}
  \text{概述（Overview）}
  \left\{
  \begin{array}{l}
    \text{AI、ML、DL}
    \left\{
    \begin{array}{l}
      \text{人工智能（Artificial Intelligence，AI）：一种思想，让计算机更像人类}\\
      \text{机器学习（Machine Learning，ML）：让计算机自主学习规律，是通往 AI 的一种途径}\\
      \text{深度学习（Deep Learning，DL）：模拟人脑，神经网络（neural network）是 ML 的一种实现方法}
    \end{array}
    \right.\\[8pt]
    \text{AI 的三要素（three elements of AI）}
    \left\{
    \begin{array}{l}
      \text{数据（data）：决定模型的上限（upper bound）}\\
      \text{算法（algorithm）}\\
      \text{算力（compute）}
      \left\{
      \begin{array}{l}
        \text{CPU（Central Processing Unit）}\\
        \text{GPU（Graphics Processing Unit）}\\
        \text{TPU（Tensor Processing Unit）}\\
        \text{昇腾（Ascend，华为 NPU）}
      \end{array}
      \right.
    \end{array}
    \right.\\[8pt]
    \text{机器学习发展史（history of ML）}
    \left\{
    \begin{array}{l}
      \text{1950 · 图灵（Alan Turing）}\\
      \text{1956 · 约翰·麦卡锡（John McCarthy），人工智能之父}\\
      \text{亚瑟·塞缪尔（Arthur Samuel），机器学习之父}\\
      \text{2012 · 辛顿（Geoffrey Hinton）、AlexNet}\\
      \text{2017 · 谷歌 Transformer 架构：编码器（encoder，理解）/ 解码器（decoder，生成）}\\
      \text{2018 · BERT（编码器）与 GPT（解码器）}\\
      \text{2022 · OpenAI 推出 ChatGPT 3.5}
    \end{array}
    \right.\\[8pt]
    \text{常用术语（terminology）}
    \left\{
    \begin{array}{l}
      \text{样本（sample）}
      \left\{
      \begin{array}{l}
        \text{一条数据}\\
        \text{包含（components）}
        \left\{
        \begin{array}{l}
          \text{特征（feature / attribute，属性）}\\
          \text{标签（label / target，目标值、预测值）}
        \end{array}
        \right.
      \end{array}
      \right.\\
      \text{数据集（dataset）}
      \left\{
      \begin{array}{l}
        \text{多个样本}\\
        \text{划分（split）}
        \left\{
        \begin{array}{l}
          \text{训练集（training set）：模型学习规律，特征 }X_{\text{train}}\text{、标签 }y_{\text{train}}\\
          \text{验证集（validation set）：调整超参数（hyperparameters，深度学习），特征 }X_{\text{val}}\text{、标签 }y_{\text{val}}\\
          \text{测试集（test set）：测试模型（深度学习）/ 模型评估（机器学习），特征 }X_{\text{test}}\text{、标签 }y_{\text{test}}\\
          \text{常见划分比例（common ratios）：8 : 1 : 1、7 : 1.5 : 1.5}
        \end{array}
        \right.
      \end{array}
      \right.
    \end{array}
    \right.\\[8pt]
    \text{算法分类（types of ML algorithms）}
    \left\{
    \begin{array}{l}
      \text{监督学习（supervised learning）：有 feature、有 label，按数据类型（data type）分}
      \left\{
      \begin{array}{l}
        \text{线性（continuous）Regression（回归）}
        \left\{
        \begin{array}{l}
          \text{Linear Regression（线性回归）：案例——波士顿房价预测（Boston housing）}\\
          \text{Polynomial Regression（多项式回归）}\\
          \text{Ridge / Lasso Regression（岭回归 / Lasso 回归）}\\
          \text{GBDT（Gradient Boosting Decision Tree）}\\
          \text{XGBoost / LightGBM}
        \end{array}
        \right.\\
        \text{离散（discrete）}\rightarrow\text{Classification（分类）}
        \left\{
        \begin{array}{l}
          \text{Logistic Regression（逻辑回归）}\\
          \text{Decision Tree（决策树）}\\
          \text{Random Forest（随机森林）}\\
          \text{KNN（K-Nearest Neighbors，K-近邻）：案例——鸢尾花分类（Iris）}\\
          \text{Naive Bayes（朴素贝叶斯）}\\
          \text{SVM（Support Vector Machine，支持向量机）}\\
          \text{XGBoost / LightGBM}
        \end{array}
        \right.
      \end{array}
      \right.\\
      \text{无监督学习（unsupervised learning）：有 feature、无 label，利用样本相似性（similarity）聚类}
      \left\{
      \begin{array}{l}
        \text{聚类（clustering）}
        \left\{
        \begin{array}{l}
          \text{K-means}\\
          \text{Hierarchical Clustering（层次聚类）}\\
          \text{DBSCAN（density-based clustering，基于密度）}
        \end{array}
        \right.\\
        \text{降维（dimensionality reduction）：PCA（Principal Component Analysis，主成分分析）}\\
        \text{关联规则（association rules）：Apriori}
      \end{array}
      \right.\\
      \text{半监督学习（semi-supervised learning）：有 feature、少 label，节省标注成本（labeling cost）}
      \left\{
      \begin{array}{l}
        \text{自训练算法（self-training）}\\
        \text{标签传播（label propagation）}\\
        \text{半监督 SVM（semi-supervised SVM）}
      \end{array}
      \right.\\
      \text{强化学习（reinforcement learning）}
      \left\{
      \begin{array}{l}
        \text{核心要素（core elements）}
        \left\{
        \begin{array}{l}
          \text{agent（智能体）}\\
          \text{环境（environment）}\\
          \text{行动（action）}\\
          \text{奖惩（reward）}
        \end{array}
        \right.\\
        \text{Q-Learning}\\
        \text{SARSA（State-Action-Reward-State-Action）}\\
        \text{DQN（Deep Q-Network）}\\
        \text{策略梯度（policy gradient）}
      \end{array}
      \right.
    \end{array}
    \right.\\[8pt]
    \text{建模流程（modeling pipeline）}
    \left\{
    \begin{array}{l}
      \text{数据准备（data preparation）}\\
      \text{数据预处理（data preprocessing）}\\
      \text{特征工程（feature engineering）}
      \left\{
      \begin{array}{l}
        \text{标准化（standardization）/ 归一化（normalization）}\\
        \text{解决量纲问题（unit scale mismatch）}
      \end{array}
      \right.\\
      \text{模型训练（model training）}\\
      \text{模型预测（prediction / inference）}\\
      \text{模型评估（model evaluation）}
    \end{array}
    \right.\\[8pt]
    \text{特征工程（feature engineering）}
    \left\{
    \begin{array}{l}
      \text{利用专业背景知识处理数据，让模型更好}\\
      \text{方式（methods）}
      \left\{
      \begin{array}{l}
        \text{特征提取（feature extraction）}\\
        \text{特征预处理（feature preprocessing）}\\
        \text{特征选择（feature selection）}\\
        \text{特征降维（dimensionality reduction）}\\
        \text{特征组合（feature crossing）}
      \end{array}
      \right.
    \end{array}
    \right.\\[8pt]
    \text{模型的拟合情况（model fit）}
    \left\{
    \begin{array}{l}
      \text{欠拟合（underfitting）}\\
      \text{正好拟合（good fit）}\\
      \text{过拟合（overfitting）}
    \end{array}
    \right.\\
    \text{环境搭建（environment setup）}
  \end{array}
  \right.
\end{array}
\right.
$$
